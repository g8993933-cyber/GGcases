            getNoise: function() {

                const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

                gain.connect(audioCtx.destination);
                osc.start(time);

            tick: function() {

        async function loadOfficialCS2Images() {
            try {
                const skinsRes = await fetch("https://raw.githubusercontent.com/ByMykel/CSGO-API/main/public/api/en/skins.json");
                if (skinsRes.ok) {
                    const apiSkins = await skinsRes.json();
                    csSkins.forEach(skin => {
                        if (skin.directImg) { skin.img = skin.directImg; return; }
                        const match = apiSkins.find(s => s.name && (s.name.toLowerCase() === (skin.search||skin.name).toLowerCase() || s.name.toLowerCase().includes((skin.search||skin.name).toLowerCase())));
                        if (match && match.image) skin.img = match.image;
                    });
                }
                const cratesRes = await fetch("https://raw.githubusercontent.com/ByMykel/CSGO-API/main/public/api/en/crates.json");
                if (cratesRes.ok) {
                    const apiCrates = await cratesRes.json();
                    const offImgs = apiCrates.filter(c => c.type === 'Case' && c.image).map(c => c.image);
                    if (offImgs.length > 0) cases.forEach((c, idx) => c.caseImg = offImgs[idx % offImgs.length]);
                }
                cases.forEach(c => c.items.forEach(it => {
                    const origin = csSkins.find(s => s.name === it.name);
                    if (origin && origin.img) it.img = origin.img;
                }));
                initCases(); renderInventory(); renderUpgrade();
            } catch (err) { console.log("Официальные изображения загружены из встроенной базы."); }
        }

        function spinCase() {
            let cost = activeCase.price * currentMultiCount;
            if (balance < cost) return alert("Недостаточно средств на балансе!");
            if (isSpinningCase) return;
            
            isSpinningCase = true;
            balance -= cost;
            stats.casesOpened += currentMultiCount;
            stats.totalWagered += cost;
            
            updateBalance();

            document.querySelectorAll('#multi-open-box .nav-btn').forEach(b => b.disabled = true);
            document.getElementById('roulette-start-btn').disabled = true;
            document.getElementById('roulette-result').innerText = '';

            const cheatProfit = document.getElementById('cheat-always-profit-case').checked;
            let availableItems = cheatProfit ? activeCase.items.filter(it => it.price >= activeCase.price) : activeCase.items;
            if(!availableItems.length) availableItems = activeCase.items;

            let results = [];
            let rtpTargetDec = targetRTP / 100;

            for (let i = 0; i < currentMultiCount; i++) {
                let r = Math.random() * availableItems.reduce((s, item) => s + item.chance, 0);
                let winItem = availableItems[0];
                for (let it of availableItems) { if (r < it.chance) { winItem = it; break; } r -= it.chance; }

                let currentRTP = stats.totalWagered > 0 ? (stats.totalWon / stats.totalWagered) : 0;

                if (!cheatProfit) {
                    if (currentRTP > rtpTargetDec && winItem.price > activeCase.price) {
                        if (Math.random() < 0.80) winItem = availableItems[0];
                    } else if (currentRTP < rtpTargetDec * 0.8 && winItem.price <= activeCase.price) {
                        const profitable = availableItems.filter(it => it.price > activeCase.price);
                        if (profitable.length > 0 && Math.random() < 0.5) winItem = profitable[Math.floor(Math.random() * profitable.length)];
                    }
                }

                stats.totalWon += winItem.price;
                checkBestItem(winItem);
                
                let finalItem = { uid: Math.random().toString(36).substr(2, 9), ...winItem };
                results.push(finalItem);

                buildTrack(i);
                const track = document.getElementById(`roulette-track-${i}`);
                const targetIndex = 32;
                const targetCard = track.children[targetIndex];
                if (targetCard) {
                    targetCard.querySelector('img').src = winItem.img;
                    targetCard.querySelector('span').innerText = winItem.name;
                }
            }

            const isFast = document.getElementById('fast-case').checked;
            const targetIndex = 32;
            const containerWidth = document.querySelector('.roulette-container').clientWidth || 600;
            const targetOffset = -(targetIndex * 138 + 69 - containerWidth / 2);

            const finalize = () => {
                let totalProfit = 0;
                results.forEach(res => {
                    inventory.push(res);
                    totalProfit += res.price;
                });

                if (currentMultiCount === 1) {
                    document.getElementById('roulette-result').innerHTML = `Выпало: <span style="color:#00f3ff; text-shadow: 0 0 8px rgba(0,243,255,0.6);">${results[0].name}</span> (${results[0].price.toLocaleString()} ₽)`;
                } else {
                    document.getElementById('roulette-result').innerHTML = `Открыто ${currentMultiCount} шт. Общий выигрыш: <span style="color:#00ff66; text-shadow: 0 0 8px rgba(0,255,102,0.6);">${totalProfit.toLocaleString()} ₽</span>`;
                }
                
                const isRare = totalProfit >= (activeCase.price * currentMultiCount * 1.5);
                if (isRare) {
                    Sound.firework();
                } else {
                    Sound.win();
                }

                saveGame();
                isSpinningCase = false;
                updateMultiOpenUI(); 
            };

            if (isFast) {
                for (let i = 0; i < currentMultiCount; i++) {
                    const track = document.getElementById(`roulette-track-${i}`);
                    track.style.transition = 'none'; 
                    track.style.transform = `translateX(${targetOffset}px)`;
                }
                finalize();
            } else {
                let ticks = 0;
                let tickTimer;
                const playTick = () => {
                    if (!isSpinningCase) return;
                    Sound.tick();
                    ticks++;
                    let nextDelay = 30 + (ticks * ticks * 0.8);
                    if (nextDelay < 600) tickTimer = setTimeout(playTick, nextDelay);
                };
                playTick();

                for (let i = 0; i < currentMultiCount; i++) {
                    const track = document.getElementById(`roulette-track-${i}`);
                    setTimeout(() => { 
                        track.style.transition = 'transform 3.8s cubic-bezier(0.08, 0.6, 0.1, 1)'; 
                        track.style.transform = `translateX(${targetOffset}px)`; 
                    }, 40 + i * 50); 
                }
                setTimeout(() => {
                    clearTimeout(tickTimer);
                    finalize();
                }, 4500);
            }
        }

        function sellItem(uid) {
            if (isUpgrading || isSpinningCase) return;
            const idx = inventory.findIndex(x => x.uid === uid);
            if (idx !== -1) {
                balance += inventory[idx].price;
                upgradeOffers = upgradeOffers.filter(x => x !== uid);
                inventory.splice(idx, 1);
                updateBalance(); renderInventory(); saveGame();
            }
        }

        function executePreset(presetStr) {
            if (isUpgrading) return;
            const offerSum = upgradeOffers.reduce((s, uid) => s + (inventory.find(x => x.uid === uid)?.price || 0), 0);
            if (offerSum === 0) return alert("Сначала выберите скины для ставки!");

            let idealPrice = (presetStr.startsWith('x') || presetStr.startsWith('X')) ? offerSum * (parseFloat(presetStr.slice(1)) || 2) : offerSum / ((parseFloat(presetStr) || 50) / 100);
            
            if (idealPrice < offerSum) idealPrice = offerSum;
            
            let closest = null, minDiff = Infinity;
            for (let i = 0; i < csSkins.length; i++) {
                if (csSkins[i].price >= offerSum) {
                    const diff = Math.abs(csSkins[i].price - idealPrice);
                    if (diff < minDiff) { minDiff = diff; closest = csSkins[i]; }
                }
            }
            if (closest) {
                upgradeTarget = closest; 
                renderUpgrade();
            }
        }

        function renderUpgrade() {
            const slotsBox = document.getElementById('slots-input'); slotsBox.innerHTML = '';
            let offerSum = 0;
            for (let i = 0; i < 6; i++) {
                if (i < upgradeOffers.length) {
                    const item = inventory.find(x => x.uid === upgradeOffers[i]);
                    if (item) {
                        offerSum += item.price;
                        slotsBox.innerHTML += `<div class="slot" style="border-color: #ff007f;"><img src="${item.img}" alt=""><button class="del-btn" onclick="removeUpgItem('${item.uid}')">×</button></div>`;
                    }
                } else slotsBox.innerHTML += `<div class="slot" onclick="openOfferModal()">+</div>`;
            }
            document.getElementById('upg-offer-sum').innerText = offerSum.toLocaleString();

            if (upgradeTarget && upgradeTarget.price < offerSum) {
                upgradeTarget = null;
            }

            const targetBox = document.getElementById('slot-target');
            if (upgradeTarget) {
                targetBox.innerHTML = `<img src="${upgradeTarget.img}" alt=""><div style="font-size: 11px; margin-top: 4px; white-space:nowrap; overflow:hidden; text-overflow:ellipsis; width:110px;">${upgradeTarget.name}</div>`;
                document.getElementById('upg-target-sum').innerText = upgradeTarget.price.toLocaleString();
            } else {
                targetBox.innerHTML = '<span style="color: #777;">+ Выбрать скин</span>'; document.getElementById('upg-target-sum').innerText = '0';
            }

            let chance = (upgradeTarget && offerSum > 0) ? Math.min(75, (offerSum / upgradeTarget.price) * 100) : 0;
            document.getElementById('upg-percent').innerText = chance.toFixed(1) + '%';
            document.getElementById('upg-run-btn').disabled = (chance === 0 || isUpgrading);

            let gradStr = "#1c142b 0deg 360deg";
            if (chance > 0) {
                const hw = chance / 2, th = [0, 25, 50, 60, 70, chance].filter(t => t <= chance);
                if (!th.includes(chance)) th.push(chance); th.sort((a,b)=>a-b);
                const colP = p => p<25?'#ff007f':p<50?'#ff7700':p<60?'#ffff00':p<70?'#00ff66':'#00f3ff';
                let rs = [], ls = [];
                th.forEach(t => { rs.push({p: (t/chance)*hw, c: colP(t)}); ls.push({p: 100 - (t/chance)*hw, c: colP(t)}); });
                ls.sort((a,b)=>a.p-b.p);
                let stops = [...rs.map(s=>`${s.c} ${s.p}%`), `#1c142b ${hw}%`, `#1c142b ${100-hw}%`, ...ls.map(s=>`${s.c} ${s.p}%`), `${rs[0].c} 100%`];
                gradStr = stops.join(', ');
            }
            document.getElementById('upg-wheel').style.background = `conic-gradient(from 180deg, ${gradStr})`;
        }

        function setArrowColor(col) { document.getElementById('upg-arrow').style.backgroundColor = col; document.getElementById('upg-arrow').style.boxShadow = `0 0 12px ${col}, 0 0 4px #fff`; }
        function openOfferModal() { if (!isUpgrading) openModal("Выберите скин", inventory.filter(x => !upgradeOffers.includes(x.uid)), it => { if (upgradeOffers.length < 6) upgradeOffers.push(it.uid); closeModal(); renderUpgrade(); }); }
        
        function openTargetModal() { 
            if (isUpgrading) return; 
            const offerSum = upgradeOffers.reduce((s, uid) => s + (inventory.find(x => x.uid === uid)?.price || 0), 0);
            
            const validSkins = csSkins.filter(s => s.price >= offerSum);
            
            if (validSkins.length === 0) {
                alert("Ваша ставка слишком высока, нет доступных скинов для апгрейда!");
                return;
            }

            openModal(`Желаемый скин (от ${offerSum.toLocaleString()} ₽)`, validSkins, it => { 
                upgradeTarget = it; 
                closeModal(); 
                renderUpgrade(); 
            }); 
        }

        function removeUpgItem(uid) { if (!isUpgrading) { upgradeOffers = upgradeOffers.filter(x => x !== uid); renderUpgrade(); } }

        function openModal(title, items, onSelect) {
            document.getElementById('modal-title').innerText = title; document.getElementById('modal-search').value = '';
            modalCurrentItems = items; modalOnSelectCallback = onSelect;
            renderModalList(items); document.getElementById('modal-select').style.display = 'flex';
        }
        function renderModalList(items) {
            const grid = document.getElementById('modal-grid'); grid.innerHTML = '';
            items.forEach(it => {
                const el = document.createElement('div'); el.className = 'card'; el.style.cursor = 'pointer';
                el.innerHTML = `<div class="card-title">${it.name}</div><img src="${it.img}" alt=""><div class="card-price">${it.price.toLocaleString()} ₽</div>`;
                el.onclick = () => modalOnSelectCallback(it); grid.appendChild(el);
            });
        }
        function filterModalSkins(q) { q = q.toLowerCase().trim(); renderModalList(!q ? modalCurrentItems : modalCurrentItems.filter(it => it.name.toLowerCase().includes(q))); }
        function closeModal() { document.getElementById('modal-select').style.display = 'none'; }

        function spinUpgrade() {
            if (isUpgrading || !upgradeTarget || !upgradeOffers.length) return;
            isUpgrading = true;
            document.getElementById('upg-run-btn').disabled = true;
            document.getElementById('upg-res-text').innerText = '';
            document.querySelectorAll('.preset-btn').forEach(b => b.disabled = true);

            const lockedTarget = { ...upgradeTarget }, lockedOffers = [...upgradeOffers];
            const offerSum = lockedOffers.reduce((s, uid) => s + (inventory.find(x => x.uid === uid)?.price || 0), 0);
            let chance = Math.min(75, (offerSum / lockedTarget.price) * 100);

            stats.totalWagered += offerSum;
            stats.upgradesDone++;

            const cheatWin = document.getElementById('cheat-always-win-upg').checked;
            let isWin = cheatWin ? true : (Math.random() * 100) <= chance;

            let currentRTP = stats.totalWagered > 0 ? (stats.totalWon / stats.totalWagered) : 0;
            let rtpTargetDec = targetRTP / 100;

            if (!cheatWin) {
                if (currentRTP > rtpTargetDec && isWin) {
                    if (Math.random() < 0.60) isWin = false;
                } else if (currentRTP < rtpTargetDec * 0.8 && !isWin) {
                    if (Math.random() < 0.40) isWin = true;
                }
            }

            if (isWin) stats.totalWon += lockedTarget.price;

            const halfWingDeg = (chance / 2) * 3.6;
            const targetAngle = isWin ? 180 + (Math.random() * (halfWingDeg * 2) - halfWingDeg) : (180 + halfWingDeg + (Math.random() * (360 - halfWingDeg * 2))) % 360;
            currentUpgRotation = Math.floor(currentUpgRotation / 360) * 360 + (5 + Math.floor(Math.random() * 3)) * 360 + targetAngle;

            const duration = document.getElementById('fast-upg').checked ? 1.5 : 3.8;
            const box = document.getElementById('upg-arrow-box');
            box.style.transition = `transform ${duration}s cubic-bezier(0.1, 0.7, 0.1, 1)`;
            box.style.transform = `rotate(${currentUpgRotation}deg)`;

            let upgTicks = 0;
            let upgTimer;
            if (!document.getElementById('fast-upg').checked) {
                const playUpgTick = () => {
                    if (!isUpgrading) return;
                    Sound.tick();
                    upgTicks++;
                    let delay = 40 + (upgTicks * 5);
                    if (delay < 400) upgTimer = setTimeout(playUpgTick, delay);
                };
                playUpgTick();
            }

            setTimeout(() => {
                clearTimeout(upgTimer);

                lockedOffers.forEach(uid => { const idx = inventory.findIndex(x => x.uid === uid); if (idx !== -1) inventory.splice(idx, 1); });
                if (isWin) {
                    Sound.firework(); 
                    inventory.push({ uid: Math.random().toString(36).substr(2, 9), ...lockedTarget });
                    document.getElementById('upg-res-text').innerHTML = '<span style="color:#00ff66; text-shadow:0 0 8px rgba(0,255,102,0.6);">УСПЕХ! Скин получен.</span>';
                    checkBestItem(lockedTarget);
                } else {
                    Sound.rimshot(); 
                    document.getElementById('upg-res-text').innerHTML = '<span style="color:#ff007f; text-shadow:0 0 8px rgba(255,0,127,0.6);">НЕУДАЧА! Предметы сгорели.</span>';
                }
                
                upgradeOffers = []; upgradeTarget = null; saveGame();
                isUpgrading = false; document.querySelectorAll('.preset-btn').forEach(b => b.disabled = false); renderUpgrade();
            }, duration * 1000 + 100);
        }

        window.onload = () => { loadGame(); initCases(); renderPresetButtons(); loadOfficialCS2Images(); };
